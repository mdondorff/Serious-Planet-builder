# SPIKE-64 scoring at about 100 m: slope p50/p90/p99 (degrees, central differences), local relief in 1 km and 5 km blocks (p50, m),
# flat share (slope under 1 degree), and the share of slopes over 20 and 33 degrees; land only (h > 0).
# Generator: the core rasters of the variants (3,072^2 at 101.8 m). Real: GLO-30 tiles block-averaged to about 100 m.
# Usage: python -I score.py <variant dir> <tiles,...> <variants,...> [real]
import sys, os, re, glob, math, json
import numpy as np
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

def metrics(a, dx, dy):
    gy, gx = np.gradient(a.astype(np.float64), dy, dx)
    s = np.degrees(np.arctan(np.hypot(gx, gy)))
    land = a > 0
    sl = s[land]
    out = {"slope": [float(np.percentile(sl, q)) for q in (50, 90, 99)], "flat": float((sl < 1).mean()),
           "over20": float((sl > 20).mean()), "over33": float((sl > 33).mean()), "land": float(land.mean())}
    for name, km in (("relief1", 1000.0), ("relief5", 5000.0)):
        by, bx = max(1, round(km / dy)), max(1, round(km / dx))
        hh, ww = (a.shape[0] // by) * by, (a.shape[1] // bx) * bx
        b = a[:hh, :ww].reshape(hh // by, by, ww // bx, bx)
        lb = land[:hh, :ww].reshape(hh // by, by, ww // bx, bx).all(axis=(1, 3))
        r = (b.max(axis=(1, 3)) - b.min(axis=(1, 3)))[lb]
        out[name] = float(np.median(r)) if r.size else float("nan")
    return out

def line(name, m):
    return f"{name:28} slope {m['slope'][0]:5.2f}/{m['slope'][1]:5.2f}/{m['slope'][2]:5.2f}  relief1 {m['relief1']:6.1f}  relief5 {m['relief5']:6.1f}  flat {100*m['flat']:5.1f} %  >20 {100*m['over20']:5.2f} %  >33 {100*m['over33']:6.3f} %"

res = {}
if len(sys.argv) > 1 and sys.argv[1] != "real":
    d, tiles, variants = sys.argv[1], sys.argv[2].split(","), sys.argv[3].split(",")
    for t in tiles:
        for v in variants:
            a = np.fromfile(f"{d}/tile{t}-{v}.f32", "<f4").reshape(3072, 3072)
            m = metrics(a, 101.8, 101.8)
            res[f"tile{t}-{v}"] = m
            print(line(f"tile{t} {v}", m), flush=True)
if sys.argv[-1] == "real":
    from glo30 import read_tile
    D = "C:/DEV/SeriousPlanetData/dem/glo30-samples/"
    names = {"N47E011": "Alps", "N48E008": "Black Forest", "N57W005": "Highlands", "N40W078": "Valley and Ridge", "N43W102": "Badlands", "N52E012": "Brandenburg", "N41W096": "Loess"}
    for f in sorted(glob.glob(D + "*.tif")):
        m = re.search(r"_(N|S)(\d+)_00_(E|W)(\d+)_00_DEM", f)
        key = "%s%02d%s%03d" % (m.group(1), int(m.group(2)), m.group(3), int(m.group(4)))
        if key not in names:
            continue
        lat = int(m.group(2))
        a, sc, tie, meta = read_tile(f)
        dy = sc[1] * 111195.0
        dx = sc[0] * 111195.0 * math.cos(math.radians(lat + 0.5))
        by, bx = max(1, round(100 / dy)), max(1, round(100 / dx))
        hh, ww = (a.shape[0] // by) * by, (a.shape[1] // bx) * bx
        b = a[:hh, :ww].reshape(hh // by, by, ww // bx, bx).mean(axis=(1, 3))
        r = metrics(b, dx * bx, dy * by)
        res["real-" + names[key]] = r
        print(line(f"real {names[key]} ({dx*bx:.0f}x{dy*by:.0f} m)", r), flush=True)
json.dump(res, open(os.environ.get("S64_JSON", "score.json"), "w"), indent=1)
