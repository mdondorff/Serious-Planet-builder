"""#154 row: lakes of the final surface (flooded, outlets edge and sea) over 1 km2 touching the 16-cell edge band; and the west lake around L (crop) x<1600."""
import sys, os, subprocess, numpy as np
from scipy import ndimage
d, exe, tags = sys.argv[1], sys.argv[2], sys.argv[3:]
n, cell = 8160, 101.802
for tag in tags:
    src = f'{d}/{tag}-final.f32'
    if not os.path.exists(src):
        np.fromfile(f'{d}/{tag}-final-8160x8160.f64', '<f8').astype('<f4').tofile(src)
    fl = f'{d}/{tag}-final-filled.f32'
    if not os.path.exists(fl):
        subprocess.run([exe, src, str(n), str(n), str(cell), '0', fl], check=True, capture_output=True)
    z = np.fromfile(src, '<f4').reshape(n, n).astype(np.float64); f = np.fromfile(fl, '<f4').reshape(n, n).astype(np.float64)
    wet = (f - z) > 0.5
    lab, k = ndimage.label(wet, structure=np.ones((3, 3)))
    ids = np.arange(1, k + 1)
    edge = np.zeros((n, n), bool); edge[:16] = edge[-16:] = True; edge[:, :16] = edge[:, -16:] = True
    box = np.zeros((n, n), bool); box[1600:-1600, 1600:-1600] = True
    area = ndimage.sum(np.ones_like(z), lab, ids) * cell * cell / 1e6
    touch = ndimage.maximum(edge.astype(float), lab, ids) > 0
    depth = ndimage.maximum(f - z, lab, ids)
    m = (area > 1) & touch
    big = area > 1
    inbox = ndimage.sum(box.astype(float), lab, ids) * cell * cell / 1e6
    order = np.argsort(-area)[:5]
    print(f'{tag} final: edge lakes over 1 km2 {int(m.sum())} ({area[m].sum():.0f} km2, depth max {depth[m].max() if m.any() else 0:.0f} m); all over 1 km2 {int(big.sum())} ({area[big].sum():.0f} km2, of it inside the box {inbox[big].sum():.0f} km2)')
    for i in order:
        ys, xs = np.nonzero(lab == ids[i])
        print(f'   lake {area[i]:.0f} km2 depth {depth[i]:.0f} m at L x {xs.min()}..{xs.max()} y {ys.min()}..{ys.max()} edge={bool(touch[i])} in-box {inbox[i]:.0f} km2')
