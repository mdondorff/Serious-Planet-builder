"""SPIKE-55 metrics: slope and local relief of a height grid, the same code for GLO-30 tiles and for the generator's windows.
Slope: central differences on the grid spacing, atan(|grad|), percentiles of the degrees. Relief: max-min over non-overlapping blocks of about 1 km and 5 km.
Descriptor: mean and standard deviation of height on a 12 x 12 grid per 1 degree cell (matches the Rust scan)."""
import json, sys, math
import numpy as np

def pct(a, qs=(50, 90, 99)):
    return [float(np.percentile(a, q)) for q in qs]

def metrics(h, dx_m, dy_m):
    h = h.astype(np.float64)
    gy, gx = np.gradient(h, dy_m, dx_m)
    slope = np.degrees(np.arctan(np.hypot(gx, gy)))
    out = {'slope_deg_p50_p90_p99': pct(slope), 'height_min': float(h.min()), 'height_max': float(h.max()), 'height_mean': float(h.mean())}
    for name, size_m in (('relief_1km', 1000.0), ('relief_5km', 5000.0)):
        bx, by = max(2, int(round(size_m / dx_m))), max(2, int(round(size_m / dy_m)))
        ny, nx = h.shape[0] // by, h.shape[1] // bx
        blocks = h[:ny * by, :nx * bx].reshape(ny, by, nx, bx)
        rel = blocks.max(axis=(1, 3)) - blocks.min(axis=(1, 3))
        out[name + '_m_p50_p90_p99'] = pct(rel)
    return out

def descriptor(h):
    """mean and std of the 12 x 12 samples per 1-degree cell, averaged over the cells the array spans (a whole tile is one cell wide)."""
    ny, nx = h.shape
    ys = ((np.arange(12) + 0.5) / 12 * ny).astype(int)
    xs = ((np.arange(12) + 0.5) / 12 * nx).astype(int)
    s = h[np.ix_(ys, xs)].astype(np.float64)
    return float(s.mean()), float(s.std())
