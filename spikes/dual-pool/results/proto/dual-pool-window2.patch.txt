+/// SPIKE-90: the chunks of synthetic background load finished in this process (the background's throughput under a knob).
+static LOAD_CHUNKS: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
+
+/// One chunk of SPIKE-90's synthetic background load (about 10 ms of hashing over the job's buffer), then the next chunk, until the system closes.
+fn load_chunk(share: Share, shared: Arc<Shared>, k: u64, buf: Arc<Mutex<Vec<u64>>>) {
+    let next = share.clone();
+    share.spawn_class(
+        Class::Idle,
+        Box::new(move || {
+            if shared.state.lock().expect("job state").closed {
+                return;
+            }
+            {
+                let mut b = buf.lock().expect("load buffer");
+                let len = b.len();
+                let mut x = 0x9E37_79B9_7F4A_7C15u64 ^ k;
+                let t = std::time::Instant::now();
+                while t.elapsed() < std::time::Duration::from_millis(10) {
+                    for _ in 0..4096 {
+                        x ^= x << 13;
+                        x ^= x >> 7;
+                        x ^= x << 17;
+                        let at = (x as usize) & (len - 1);
+                        b[at] = b[at].wrapping_add(x);
+                    }
+                }
+                std::hint::black_box(&*b);
+            }
+            LOAD_CHUNKS.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
+            load_chunk(next, shared, k.wrapping_add(1), buf);
+        }),
+    );
