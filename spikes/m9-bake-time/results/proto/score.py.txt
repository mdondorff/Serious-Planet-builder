# SPIKE-70 scoring: SPIKE-65's measures (slope p50/p90/p99, 1 km and 5 km relief p50, flat share, shares over 20 and 33 degrees) on the
# central 3,072^2 of each 6,656^2 result (round the four-tile corner), and the variant over the joint bake per measure.
# Usage: python -I score.py <dir> <sites> <variants> <real.json> <out.json>
import sys, json
import numpy as np
d, sites, variants, realp, outp = sys.argv[1], sys.argv[2].split(','), sys.argv[3].split(','), sys.argv[4], sys.argv[5]
B, C0, C1 = 6656, 1792, 4864

def metrics(a, dx=101.8):
    gy, gx = np.gradient(a.astype(np.float64), dx, dx)
    s = np.degrees(np.arctan(np.hypot(gx, gy)))
    land = a > 0
    sl = s[land]
    out = {"slope": [float(np.percentile(sl, q)) for q in (50, 90, 99)], "flat": float((sl < 1).mean()),
           "over20": float((sl > 20).mean()), "over33": float((sl > 33).mean())}
    for name, km in (("relief1", 1000.0), ("relief5", 5000.0)):
        by = max(1, round(km / dx))
        hh = (a.shape[0] // by) * by
        b = a[:hh, :hh].reshape(hh // by, by, hh // by, by)
        lb = land[:hh, :hh].reshape(hh // by, by, hh // by, by).all(axis=(1, 3))
        r = (b.max(axis=(1, 3)) - b.min(axis=(1, 3)))[lb]
        out[name] = float(np.median(r)) if r.size else float("nan")
    return out

real = json.load(open(realp))
RN = {'alps': 'real-Alps', 'blackforest': 'real-Black Forest', 'highlands': 'real-Highlands', 'valleyridge': 'real-Valley and Ridge'}
res = {}
for s in sites:
    for v in variants:
        path = f'{d}/{s}-{v}.f32' if v != 's68tiles' else f'{d}/../relief68/out/{s}-hard.f32'
        try:
            full = np.memmap(path, dtype='<f4', mode='r', shape=(B, B))
        except FileNotFoundError:
            continue
        m = metrics(np.array(full[C0:C1, C0:C1]))
        # Seam: the surface's own step across the core edges (row/column 3,327 to 3,328) against its steps at 150 and 300 cells off the edge, p99 of each, land only.
        def steps(off):
            x = 3328 + off
            col = np.abs(full[256:6400, x].astype(np.float64) - full[256:6400, x - 1])
            row = np.abs(full[x, 256:6400].astype(np.float64) - full[x - 1, 256:6400])
            land = np.concatenate([(full[256:6400, x] > 0) & (full[256:6400, x - 1] > 0), (full[x, 256:6400] > 0) & (full[x - 1, 256:6400] > 0)])
            return np.concatenate([col, row])[land]
        e = steps(0)
        r = np.concatenate([steps(o) for o in (-300, -150, 150, 300)])
        m['seam_p99'] = float(np.percentile(e, 99)); m['ref_p99'] = float(np.percentile(r, 99)); m['seam_p50'] = float(np.percentile(e, 50)); m['ref_p50'] = float(np.percentile(r, 50))
        res[f'{s}-{v}'] = m
    for v in variants:
        m = res.get(f'{s}-{v}')
        if not m:
            continue
        j = res.get(f'{s}-joint')
        r = real.get(RN[s], {})
        rel = lambda k, ref: m[k] / ref[k] if ref and ref.get(k) else float('nan')
        print(f"{s:12s} {v:12s} slope {m['slope'][0]:5.2f}/{m['slope'][1]:5.2f}/{m['slope'][2]:5.2f} relief1 {m['relief1']:6.1f} relief5 {m['relief5']:6.1f} flat {100*m['flat']:4.1f} % >20 {100*m['over20']:5.2f} % >33 {100*m['over33']:5.2f} %"
              f" | /joint relief1 {rel('relief1', j):.2f} relief5 {rel('relief5', j):.2f} slope50 {m['slope'][0]/j['slope'][0]:.2f}"
              f" | seam p50/p99 {m['seam_p50']:.1f}/{m['seam_p99']:.1f} vs off-edge {m['ref_p50']:.1f}/{m['ref_p99']:.1f}"
              f" | /real relief1 {rel('relief1', r):.2f} relief5 {rel('relief5', r):.2f} slope50 {m['slope'][0]/r['slope'][0]:.2f}")
json.dump(res, open(outp, 'w'), indent=1)
