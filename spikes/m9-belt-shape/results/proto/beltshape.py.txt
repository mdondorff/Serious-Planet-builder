# SPIKE-79: belts, not blobs. A belt-shaped range mask in place of the blob-shaped range zone, on SPIKE-69's map of the generator's inputs
# (720 x 360 x 13 f32), resampled bilinearly to 0.25 degree. Today's belt term (SPIKE-78's amplitude 7,000 m) is `A ridge^2 zone inland`;
# the proposal is `A B (c + (1 - c) ridge^2)` with a belt mask B:
# - interior sutures: the zero lines of a smooth fBm (long, continuous, curved), a Gaussian cross-strike profile of the distance to the line,
#   cut into long segments by a second, lower-frequency fBm;
# - coastal arcs (Andean margins): the line at a fixed land depth inland of the coast, on part of the coasts (a third fBm picks the active margins);
# - gated at the coast only (not by `inland`), so a coastal arc can reach the sea.
# SPIKE-77's plateau rule is kept, away from B instead of the range zone. Scores: hypsometry (SPIKE-78's targets) and the shape of land above 2 km
# (connected regions, longitude wrap; Feret length L, area A, aspect L^2 / A = L / W for a band).
# Usage: python -I beltshape.py <map13.f32> <out dir> [variant ...]
import sys, itertools, json, time, os
import numpy as np
from PIL import Image
from scipy import ndimage

mp, out = sys.argv[1], sys.argv[2]
t0 = time.time()
R = 6371.0
a = np.fromfile(mp, dtype='<f4').reshape(360, 720, 13).astype(np.float64)
# Resample to 0.25 degree (bilinear, longitude wraps).
H, W = 720, 1440
lat = 90 - (np.arange(H) + 0.5) * 0.25
lon = -180 + (np.arange(W) + 0.5) * 0.25
fy = np.clip((90 - lat) / 0.5 - 0.5, 0, 359)
fx = (lon + 180) / 0.5 - 0.5
y0 = np.minimum(np.floor(fy).astype(int), 358)
ty = (fy - y0)[:, None]
x0 = np.floor(fx).astype(int) % 720
x1 = (x0 + 1) % 720
tx = (fx - np.floor(fx))[None, :]


def up(ch):
    c = a[..., ch]
    c = np.nan_to_num(c)
    r0 = c[y0][:, x0] * (1 - tx) + c[y0][:, x1] * tx
    r1 = c[y0 + 1][:, x0] * (1 - tx) + c[y0 + 1][:, x1] * tx
    return r0 * (1 - ty) + r1 * ty


cont, zone, inland, ridge, hill, e = [up(i) for i in (2, 3, 4, 5, 6, 7)]
LAT = np.repeat(lat[:, None], W, 1)
LON = np.repeat(lon[None, :], H, 0)
landall = cont > 0.1
land = landall & (np.abs(LAT) < 60)
cell_km2 = (R * np.radians(0.25)) ** 2 * np.cos(np.radians(LAT))
wgt = cell_km2 * land
base = e - 5200 * ridge * zone * inland - hill
ld = cont - 0.1


def smooth(x, lo, hi):
    t = np.clip((x - lo) / (hi - lo), 0, 1)
    return t * t * (3 - 2 * t)


def vnoise(p, freq, seed):
    q = p * freq
    i = np.floor(q).astype(np.int64)
    f = q - i
    t = f * f * (3 - 2 * f)
    def h(x, y, z):
        v = (x * 73856093) ^ (y * 19349663) ^ (z * 83492791) ^ seed
        v = (v ^ (v >> 13)) * 1274126177
        v = v ^ (v >> 16)
        return (v & 0xffff) / 32767.5 - 1.0
    acc = 0
    for dx, dy, dz in itertools.product((0, 1), repeat=3):
        wx = t[..., 0] if dx else 1 - t[..., 0]
        wy = t[..., 1] if dy else 1 - t[..., 1]
        wz = t[..., 2] if dz else 1 - t[..., 2]
        acc = acc + wx * wy * wz * h(i[..., 0] + dx, i[..., 1] + dy, i[..., 2] + dz)
    return acc


def fbm(p, freq, oct, seed):
    s, amp, n = 0, 1.0, 0
    for o in range(oct):
        s = s + amp * vnoise(p, freq * 2 ** o, seed + o)
        n += amp
        amp *= 0.5
    return s / n


la, lo = np.radians(LAT), np.radians(LON)
P3 = np.stack([np.cos(la) * np.cos(lo), np.cos(la) * np.sin(lo), np.sin(la)], -1)


def grad_km(f):
    # |grad f| per km on the lat/lon grid (longitude wraps).
    dx = (np.roll(f, -1, 1) - np.roll(f, 1, 1)) / (2 * R * np.radians(0.25) * np.maximum(np.cos(np.radians(LAT)), 0.05))
    dy = np.zeros_like(f)
    dy[1:-1] = (f[:-2] - f[2:]) / (2 * R * np.radians(0.25))
    return np.hypot(dx, dy) + 1e-12


def line_profile(f, half_width_km):
    d = np.abs(f) / grad_km(f)
    return np.exp(-(d / half_width_km) ** 2)


# SPIKE-77's plateau noise.
pn = 0.5 + 0.5 * fbm(P3, 1.6, 3, 4242)


def elevation(B, amp, c, plateau_away):
    belt = amp * B * (c + (1 - c) * ridge)  # ridge is already squared in the map
    P = smooth(pn, 0.50, 0.62) * smooth(inland, 0.2, 0.6) * (1 - smooth(plateau_away, 0.2, 0.6))
    return np.clip(base + belt + hill + np.where(landall, 1200 * P, 0), -9000, 9000)


EARTH = {2000: 0.0666, 3000: 0.0315, 4000: 0.0194, 5000: 0.0077}


def hyps(ev):
    s = {t: float((wgt * (ev >= t)).sum() / wgt.sum()) for t in (1000, 2000, 3000, 4000, 5000)}
    err = sum(abs(np.log(max(s[t], 1e-5) / EARTH[t])) for t in EARTH)
    return s, err


def regions(ev, thr=2000.0, min_km2=50000.0):
    m = land & (ev >= thr)
    lab, n = ndimage.label(m, structure=np.ones((3, 3)))
    # Longitude wrap: merge labels touching across the seam.
    parent = list(range(n + 1))
    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x
    for r in range(H):
        for dr in (-1, 0, 1):
            rr = r + dr
            if 0 <= rr < H and lab[r, 0] and lab[rr, W - 1]:
                parent[find(lab[r, 0])] = find(lab[rr, W - 1])
    roots = np.array([find(i) for i in range(n + 1)])
    lab = roots[lab]
    out = []
    rng = np.random.default_rng(1)
    for k in np.unique(lab[lab > 0]):
        idx = np.nonzero(lab == k)
        area = float(cell_km2[idx].sum())
        if area < min_km2:
            continue
        v = P3[idx]
        if len(v) > 1500:
            v = v[rng.choice(len(v), 1500, replace=False)]
        dots = np.clip(v @ v.T, -1, 1)
        L = float(R * np.arccos(dots.min()))
        out.append({'area_km2': area, 'length_km': L, 'width_km': area / L, 'aspect': L * L / area, 'lat': float(LAT[idx].mean())})
    out.sort(key=lambda r: -r['area_km2'])
    return out


def shape_summary(rs):
    if not rs:
        return {'regions': 0}
    A = np.array([r['area_km2'] for r in rs])
    asp = np.array([r['aspect'] for r in rs])
    o = np.argsort(asp)
    c = np.cumsum(A[o]) / A.sum()
    return {'regions': len(rs), 'aspect_median': float(np.median(asp)), 'aspect_area_weighted_median': float(asp[o][np.searchsorted(c, 0.5)]),
            'area_share_aspect_ge_4': float(A[asp >= 4].sum() / A.sum()), 'longest_km': float(max(r['length_km'] for r in rs))}


# The belt masks.
# Knobs (environment): line frequency, segment gates (lower edge; the window is 0.14 wide) and the coastal line's land depth.
FI = float(os.environ.get('S79_FI', '2.2'))
GI = float(os.environ.get('S79_GI', '0.35'))
GC = float(os.environ.get('S79_GC', '0.40'))
DC = float(os.environ.get('S79_DC', '0.10'))
n_int = fbm(P3, FI, 3, 7901)
seg_int = smooth(0.5 + 0.5 * fbm(P3, 1.1, 2, 7911), GI, GI + 0.14)
n_coast = ld - DC
seg_coast = smooth(0.5 + 0.5 * fbm(P3, 0.9, 2, 7921), GC, GC + 0.12)
coast_gate = smooth(ld, 0.0, 0.04)


# Plates mode (S79_MODE=plates): the interior belts are the convergent margins of a spherical Voronoi tessellation of S79_NP plates, each with a
# random Euler-pole rotation; the direction is warped a little (0.12) so the margins curve. The distance to a margin is half the angular
# difference between the nearest and the second-nearest plate centre (exact on the bisector), and the convergence is the relative velocity
# against the margin normal, as a smoothstep gate.
MODE = os.environ.get('S79_MODE', 'lines')
if MODE == 'plates':
    NP = int(os.environ.get('S79_NP', '14'))
    rg = np.random.default_rng(7931)
    cen = rg.normal(size=(NP, 3))
    cen /= np.linalg.norm(cen, axis=1, keepdims=True)
    poles = rg.normal(size=(NP, 3))
    poles /= np.linalg.norm(poles, axis=1, keepdims=True)
    rate = rg.uniform(0.3, 1.0, NP)
    wp = np.stack([fbm(P3 + o, 1.3, 2, 7941 + k) for k, o in enumerate((0.0, 11.0, 23.0))], -1)
    Q = P3 + float(os.environ.get('S79_WARP', '0.12')) * wp
    Q /= np.linalg.norm(Q, axis=-1, keepdims=True)
    ang = np.arccos(np.clip(Q @ cen.T, -1, 1))
    order = np.argsort(ang, axis=-1)
    i1, i2 = order[..., 0], order[..., 1]
    a1 = np.take_along_axis(ang, i1[..., None], -1)[..., 0]
    a2 = np.take_along_axis(ang, i2[..., None], -1)[..., 0]
    d_margin_km = 0.5 * (a2 - a1) * R
    vel = lambda i: np.cross((poles * rate[:, None])[i], P3)
    nrm = cen[i2] - cen[i1]
    nrm = nrm - (nrm * P3).sum(-1, keepdims=True) * P3
    nrm /= np.linalg.norm(nrm, axis=-1, keepdims=True) + 1e-12
    conv = ((vel(i1) - vel(i2)) * nrm).sum(-1)  # plate 1 moving towards plate 2 is positive
    conv_gate = smooth(conv, 0.05, 0.35)


def belt_mask(hw_int, hw_coast, coast_on=True):
    if MODE == 'plates':
        B = conv_gate * np.exp(-(d_margin_km / hw_int) ** 2)
    else:
        B = seg_int * line_profile(n_int, hw_int)
    if coast_on:
        B = np.maximum(B, seg_coast * line_profile(n_coast, hw_coast))
    return B * coast_gate


variants = {}
variants['today (zone blobs, 7000 m)'] = (np.clip(base + 7000 * ridge * zone * inland + hill + np.where(landall, 1200 * smooth(pn, 0.50, 0.62) * smooth(inland, 0.2, 0.6) * (1 - smooth(zone, 0.2, 0.6)), 0), -9000, 9000), zone * inland)
res = {'variants': {}}
t_mask = time.time()
grid = []
for hw in (70.0, 90.0, 110.0):
    B = belt_mask(hw, hw * 0.8)
    for c in (0.35, 0.5, 0.65):
        for amp in (5000.0, 5500.0, 6000.0, 6500.0, 7000.0):
            ev = elevation(B, amp, c, B)
            s, err = hyps(ev)
            grid.append((err, hw, c, amp, s))
grid.sort(key=lambda g: g[0])
res['grid'] = [{'err': g[0], 'half_width_km': g[1], 'c': g[2], 'amp': g[3], 'above': {str(k): v for k, v in g[4].items()}} for g in grid[:10]]
print(f'knobs FI {FI} GI {GI} GC {GC} DC {DC}; mask land share (B>0.5) {float((wgt*(belt_mask(130.0,104.0)>0.5)).sum()/wgt.sum()):.3f}, zone inland>0.5 {float((wgt*(zone*inland>0.5)).sum()/wgt.sum()):.3f}')
print('grid best (hypsometry):')
for g in grid[:6]:
    s = g[4]
    print(f"  hw {g[1]:.0f} km c {g[2]} amp {g[3]:.0f}: 1/2/3/4/5 km {s[1000]:.3f}/{s[2000]:.3f}/{s[3000]:.3f}/{s[4000]:.3f}/{s[5000]:.4f} err {g[0]:.2f}")
# Shape for the three best hypsometry settings with distinct half-widths, plus the best overall.
chosen = []
seen = set()
for g in grid:
    if g[1] not in seen:
        seen.add(g[1])
        chosen.append(g)
for g in chosen:
    B = belt_mask(g[1], g[1] * 0.8)
    variants[f'belts hw {g[1]:.0f} km, c {g[2]}, amp {g[3]:.0f}'] = (elevation(B, g[3], g[2], B), B)
Bn = belt_mask(chosen[0][1], chosen[0][1] * 0.8, coast_on=False)  # noqa
variants[f'belts without coastal arcs (hw {chosen[0][1]:.0f})'] = (elevation(Bn, chosen[0][3], chosen[0][2], Bn), Bn)
for name, (ev, B) in variants.items():
    s, err = hyps(ev)
    rs = regions(ev)
    sm = shape_summary(rs)
    res['variants'][name] = {'above': {str(k): v for k, v in s.items()}, 'err': err, 'shape': sm, 'regions': rs[:12], 'max': float(ev[land].max())}
    print(f"{name:48s} | 1/2/3/4/5 km {s[1000]:.3f}/{s[2000]:.3f}/{s[3000]:.3f}/{s[4000]:.3f}/{s[5000]:.4f} err {err:.2f} | max {ev[land].max():.0f} | "
          f"regions {sm.get('regions')} aspect median {sm.get('aspect_median', 0):.1f} area-weighted {sm.get('aspect_area_weighted_median', 0):.1f} "
          f"share aspect>=4 {sm.get('area_share_aspect_ge_4', 0):.2f} longest {sm.get('longest_km', 0):.0f} km")
    for r in rs[:6]:
        print(f"    region {r['area_km2']/1e3:7.0f} k km2 | L {r['length_km']:5.0f} km | W {r['width_km']:4.0f} km | aspect {r['aspect']:5.1f} | lat {r['lat']:.0f}")
res['seconds'] = time.time() - t0
json.dump(res, open(f'{out}/beltshape.json', 'w'), indent=1)

# Images: elevation 0-6 km (today | belts) and the masks (zone inland | B), 0.25 degree, downsampled 2x for the phone.
def ramp(x):
    t = np.clip(x / 6000.0, 0, 1)[..., None]
    lo_c, mid_c, hi_c = np.array([0.30, 0.55, 0.25]), np.array([0.65, 0.45, 0.25]), np.array([1.0, 1.0, 1.0])
    c = np.where(t < 0.5, lo_c + (mid_c - lo_c) * (t / 0.5), mid_c + (hi_c - mid_c) * ((t - 0.5) / 0.5))
    return np.where(landall[..., None], c, np.array([0.12, 0.2, 0.35]))
def gray(x):
    return np.where(landall[..., None], np.clip(x, 0, 1)[..., None] * np.ones(3), np.array([0.12, 0.2, 0.35]))
names = list(variants)
best = next((n for n in names if n.startswith(os.environ.get('S79_PICK', 'belts hw 90 km'))), names[1])
def save(img, path):
    im = Image.fromarray((img * 255).astype(np.uint8)).resize((img.shape[1] // 2, img.shape[0] // 2), Image.LANCZOS)
    im.save(path)
sep = np.ones((H, 8, 3))
save(np.concatenate([ramp(variants[names[0]][0]), sep, ramp(variants[best][0])], 1), f'{out}/elevation-blobs-vs-belts.png')
save(np.concatenate([gray(variants[names[0]][1]), sep, gray(variants[best][1])], 1), f'{out}/mask-zone-vs-belt.png')
def above2(ev):
    return np.where(landall[..., None], np.where((ev >= 2000)[..., None], np.array([0.85, 0.3, 0.2]), np.array([0.62, 0.72, 0.5])), np.array([0.12, 0.2, 0.35]))
save(np.concatenate([above2(variants[names[0]][0]), sep, above2(variants[best][0])], 1), f'{out}/above-2km-blobs-vs-belts.png')
print('images for', best, f'| {time.time() - t0:.0f} s')
