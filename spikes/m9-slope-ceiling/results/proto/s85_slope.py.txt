"""SPIKE-85: slope distribution and the share of 1 km relief carried by steep slopes, GLO-30 tiles.
Slope: atan of the central-difference gradient on a square grid of spacing s (block means of the 1 arc-second tile).
Steep share of relief: the slope-limited surface m = (lower + upper envelope)/2, where lower = min_y(h(y) + t|x-y|) and
upper = max_y(h(y) - t|x-y|), t = tan(ceiling); m has slope <= t (8-neighbour chamfer metric) and equals h where h already does.
The relief p50 of m against h says how much of the 1 km relief needs slopes above the ceiling."""
import json, math, re, sys
import numpy as np
from glo30 import read_tile

SITES = {'N47E011': 'Alps (Tyrol)', 'N57W005': 'Scottish Highlands', 'N48E008': 'Black Forest', 'N40W078': 'Valley and Ridge', 'N27E086': 'Himalaya (Everest), reference'}
SPACINGS = (30, 60, 100, 200, 600)
CEILS = (35, 45, 55, 65)

def resample(h, dx, dy, s):
    fy, fx = max(1, round(s / dy)), max(1, round(s / dx))
    ny, nx = h.shape[0] // fy, h.shape[1] // fx
    b = h[:ny * fy, :nx * fx].reshape(ny, fy, nx, fx).mean(axis=(1, 3))
    return b, dx * fx, dy * fy

def slope_deg(h, dx, dy):
    gy, gx = np.gradient(h, dy, dx)
    return np.degrees(np.arctan(np.hypot(gx, gy)))

def row_pass(r, c):
    # min over k of r[k] + |j-k| c, both directions, vectorised
    j = np.arange(r.size) * c
    a = np.minimum.accumulate(r - j) + j
    b = (np.minimum.accumulate((r + j)[::-1]))[::-1] - j
    return np.minimum(a, b)

def lower_env(h, dx, dy, t):
    h = h.copy(); cx, cy, cd = dx * t, dy * t, math.hypot(dx, dy) * t
    for _ in range(2):
        for order in (range(h.shape[0]), range(h.shape[0] - 1, -1, -1)):
            prev = None
            for i in order:
                r = h[i]
                if prev is not None:
                    c = prev + cy
                    c[1:] = np.minimum(c[1:], prev[:-1] + cd)
                    c[:-1] = np.minimum(c[:-1], prev[1:] + cd)
                    r = np.minimum(r, c)
                r = row_pass(r, cx)
                h[i] = r; prev = r
    return h

def relief_p50(h, dx, dy, size=1000.0):
    bx, by = max(2, int(round(size / dx))), max(2, int(round(size / dy)))
    ny, nx = h.shape[0] // by, h.shape[1] // bx
    blk = h[:ny * by, :nx * bx].reshape(ny, by, nx, bx)
    return float(np.percentile(blk.max(axis=(1, 3)) - blk.min(axis=(1, 3)), 50))

def main(dem_dir, out_json):
    res = {}
    for key, name in SITES.items():
        m = re.match(r'([NS])(\d+)([EW])(\d+)', key)
        lat = int(m.group(2)) * (1 if m.group(1) == 'N' else -1)
        f = '%s/Copernicus_DSM_COG_10_%s%02d_00_%s%03d_00_DEM.tif' % (dem_dir, m.group(1), int(m.group(2)), m.group(3), int(m.group(4)))
        a, sc, tie, meta = read_tile(f)
        a = a.astype(np.float64)
        dy0 = sc[1] * 111195.0; dx0 = sc[0] * 111195.0 * math.cos(math.radians(lat + 0.5))
        r = {'landform': name, 'native_dx_m': dx0, 'native_dy_m': dy0, 'by_spacing': {}}
        for s in SPACINGS:
            h, dx, dy = resample(a, dx0, dy0, s)
            sl = slope_deg(h, dx, dy)
            row = {'dx_m': dx, 'dy_m': dy, 'slope_p50_p90_p99_max': [float(np.percentile(sl, q)) for q in (50, 90, 99)] + [float(sl.max())],
                   'share_above': {str(c): float((sl > c).mean()) for c in CEILS}}
            if s in (30, 100):
                rp = relief_p50(h, dx, dy)
                row['relief1km_p50'] = rp
                row['relief1km_p50_ratio_limited'] = {}
                for c in CEILS:
                    t = math.tan(math.radians(c))
                    lo = lower_env(h, dx, dy, t); up = -lower_env(-h, dx, dy, t)
                    row['relief1km_p50_ratio_limited'][str(c)] = relief_p50(0.5 * (lo + up), dx, dy) / rp
                    row.setdefault('relief1km_p50_ratio_lower', {})[str(c)] = relief_p50(lo, dx, dy) / rp
            r['by_spacing'][str(s)] = row
            print(key, s, 'm', 'slope p50/p90/p99/max %.1f/%.1f/%.1f/%.1f' % tuple(row['slope_p50_p90_p99_max']),
                  'above', ' '.join('%s:%.3f' % kv for kv in row['share_above'].items()),
                  ('relief %.0f ratio ' % row['relief1km_p50'] + ' '.join('%s:%.2f' % kv for kv in row['relief1km_p50_ratio_limited'].items())) if 'relief1km_p50' in row else '', flush=True)
        res[key] = r
    json.dump(res, open(out_json, 'w'), indent=1)

if __name__ == '__main__':
    main(sys.argv[1], sys.argv[2])
