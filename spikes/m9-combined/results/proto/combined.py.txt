# SPIKE-80: the re-derived generator, combined, at coarse global resolution (0.25 degree, resampled from SPIKE-69's 0.5-degree map of the generator's
# inputs, 720 x 360 x 13 f32). The base elevation keeps today's continents and coast ramp (`base`, 0-1,000 m); everything above it is new:
# - belts: SPIKE-79's plate-margin mask (14 plates, convergent margins, warp 0.35, coastal arcs) with a half-width varying along strike
#   (55-135 km), belt term `A B (c + (1 - c) ridge^2)`;
# - back-arc plateaus: behind the strongest convergent margins, on the plate with more land, a flat-topped uplift `U` out to 250-600 km;
# - old margins (the non-convergent ones, the worn-down orogens): a hill belt (half-width 150 km, uplift 350 m);
# - SPIKE-77's plateau rule (1,200 m), away from belts and back-arc plateaus;
# - the hill band is dropped (SPIKE-76: class amplitude replaces it).
# The default class map: high mountains (belt core above 2.5 km), mountains (belt core), hills (belt flanks and old margins), plateau, rolling
# plain, plain; class k as SPIKE-76 (1 / 1 / 5 / 6 / 7.5, plateau 1) blended by memberships. Scores: hypsometry (SPIKE-78) and class shares
# (SPIKE-69's relief classes, SPIKE-77's method `k R14` and the class membership). Writes the delta elevation (new minus today's sample) and the
# class fields for the site runs, re-matches the scorecard sites by class and mean height, and draws the maps.
# Usage: python -I combined.py <map13.f32> <out dir> <real stats json or ->
import sys, itertools, json, time, os
import numpy as np
from PIL import Image

mp, out = sys.argv[1], sys.argv[2]
t0 = time.time()
R = 6371.0
a = np.fromfile(mp, dtype='<f4').reshape(360, 720, 13).astype(np.float64)
H, W = 720, 1440
lat = 90 - (np.arange(H) + 0.5) * 0.25
lon = -180 + (np.arange(W) + 0.5) * 0.25
fy = np.clip((90 - lat) / 0.5 - 0.5, 0, 359)
fx = (lon + 180) / 0.5 - 0.5
y0 = np.minimum(np.floor(fy).astype(int), 358)
ty = (fy - y0)[:, None]
x0 = np.floor(fx).astype(int) % 720
x1 = (x0 + 1) % 720
tx = (fx - np.floor(fx))[None, :]


def up(ch):
    c = np.nan_to_num(a[..., ch])
    r0 = c[y0][:, x0] * (1 - tx) + c[y0][:, x1] * tx
    r1 = c[y0 + 1][:, x0] * (1 - tx) + c[y0 + 1][:, x1] * tx
    return r0 * (1 - ty) + r1 * ty


cont, zone, inland, ridge, hill, e, s50, r14 = [up(i) for i in (2, 3, 4, 5, 6, 7, 9, 12)]
LAT = np.repeat(lat[:, None], W, 1)
LON = np.repeat(lon[None, :], H, 0)
landall = cont > 0.1
land = landall & (np.abs(LAT) < 60)
cell_km2 = (R * np.radians(0.25)) ** 2 * np.cos(np.radians(LAT))
wgt = cell_km2 * land
base = e - 5200 * ridge * zone * inland - hill
ld = cont - 0.1


def smooth(x, lo, hi):
    t = np.clip((x - lo) / (hi - lo), 0, 1)
    return t * t * (3 - 2 * t)


def vnoise(p, freq, seed):
    q = p * freq
    i = np.floor(q).astype(np.int64)
    f = q - i
    t = f * f * (3 - 2 * f)
    def h(x, y, z):
        v = (x * 73856093) ^ (y * 19349663) ^ (z * 83492791) ^ seed
        v = (v ^ (v >> 13)) * 1274126177
        v = v ^ (v >> 16)
        return (v & 0xffff) / 32767.5 - 1.0
    acc = 0
    for dx, dy, dz in itertools.product((0, 1), repeat=3):
        wx = t[..., 0] if dx else 1 - t[..., 0]
        wy = t[..., 1] if dy else 1 - t[..., 1]
        wz = t[..., 2] if dz else 1 - t[..., 2]
        acc = acc + wx * wy * wz * h(i[..., 0] + dx, i[..., 1] + dy, i[..., 2] + dz)
    return acc


def fbm(p, freq, oct, seed):
    s, amp, n = 0, 1.0, 0
    for o in range(oct):
        s = s + amp * vnoise(p, freq * 2 ** o, seed + o)
        n += amp
        amp *= 0.5
    return s / n


la, lo = np.radians(LAT), np.radians(LON)
P3 = np.stack([np.cos(la) * np.cos(lo), np.cos(la) * np.sin(lo), np.sin(la)], -1)


def grad_km(f):
    dx = (np.roll(f, -1, 1) - np.roll(f, 1, 1)) / (2 * R * np.radians(0.25) * np.maximum(np.cos(np.radians(LAT)), 0.05))
    dy = np.zeros_like(f)
    dy[1:-1] = (f[:-2] - f[2:]) / (2 * R * np.radians(0.25))
    return np.hypot(dx, dy) + 1e-12


# Plates (SPIKE-79, mode `plates`, 14 plates, warp 0.35).
NP = 14
rg = np.random.default_rng(7931)
cen = rg.normal(size=(NP, 3))
cen /= np.linalg.norm(cen, axis=1, keepdims=True)
poles = rg.normal(size=(NP, 3))
poles /= np.linalg.norm(poles, axis=1, keepdims=True)
rate = rg.uniform(0.3, 1.0, NP)
wp = np.stack([fbm(P3 + o, 1.3, 2, 7941 + k) for k, o in enumerate((0.0, 11.0, 23.0))], -1)
Q3 = P3 + 0.35 * wp
Q3 /= np.linalg.norm(Q3, axis=-1, keepdims=True)
ang = np.arccos(np.clip(Q3 @ cen.T, -1, 1))
order = np.argsort(ang, axis=-1)
i1, i2 = order[..., 0], order[..., 1]
a1 = np.take_along_axis(ang, i1[..., None], -1)[..., 0]
a2 = np.take_along_axis(ang, i2[..., None], -1)[..., 0]
d_margin = 0.5 * (a2 - a1) * R
vel = lambda i: np.cross((poles * rate[:, None])[i], P3)
nrm = cen[i2] - cen[i1]
nrm = nrm - (nrm * P3).sum(-1, keepdims=True) * P3
nrm /= np.linalg.norm(nrm, axis=-1, keepdims=True) + 1e-12
conv = ((vel(i1) - vel(i2)) * nrm).sum(-1)
conv_gate = smooth(conv, 0.05, 0.35)
old_gate = 1 - smooth(conv, -0.05, 0.15)  # the non-convergent margins: worn-down orogens
# The plate with more land overrides: its side of a strong margin gets the back-arc plateau.
landfrac = np.array([float((wgt * (i1 == k) * landall).sum() / max((cell_km2 * (i1 == k)).sum(), 1)) for k in range(NP)])
over = landfrac[i1] >= landfrac[i2]
coast_gate = smooth(ld, 0.0, 0.04)
seg_coast = smooth(0.5 + 0.5 * fbm(P3, 0.9, 2, 7921), 0.40, 0.52)
n_coast = ld - 0.10
d_coast = np.abs(n_coast) / grad_km(n_coast)
pn = 0.5 + 0.5 * fbm(P3, 1.6, 3, 4242)
hw_var = 0.5 + 0.5 * fbm(P3, 2.5, 2, 8011)  # along-strike width noise in [0, 1]

KN = {k: float(os.environ.get('S80_' + k, v)) for k, v in dict(A='6000', C='0.5', HW='90', WV='1', UBA='3000', BAW='450', BAS='0.45', HILLU='350').items()}


def model(A, c, hw0, width_var, Uba, ba_w, ba_strength, hill_u):
    hw = hw0 * ((0.6 + 0.9 * hw_var) if width_var else 1.0)
    core = np.maximum(conv_gate * np.exp(-(d_margin / hw) ** 2), seg_coast * np.exp(-(d_coast / (0.8 * hw)) ** 2)) * coast_gate
    flank = np.maximum(conv_gate * np.exp(-(d_margin / (3.0 * hw)) ** 2), seg_coast * np.exp(-(d_coast / (2.4 * hw)) ** 2)) * coast_gate
    oldm = old_gate * np.exp(-(d_margin / 150.0) ** 2) * coast_gate
    # Back-arc: behind strong margins (convergence well above the gate), flat top out to ba_w km, on the overriding side.
    strong = smooth(conv, ba_strength, ba_strength + 0.25)
    ba = strong * over * (1 - smooth(d_margin, ba_w, ba_w + 250.0)) * smooth(inland, 0.1, 0.5) * coast_gate
    belt = A * core * (c + (1 - c) * ridge)
    hills = hill_u * np.maximum(oldm, 0.5 * flank)
    P = smooth(pn, 0.50, 0.62) * smooth(inland, 0.2, 0.6) * (1 - smooth(np.maximum(flank, ba), 0.2, 0.6))
    ev = np.clip(base + belt + Uba * ba + hills + np.where(landall, 1200 * P, 0), -9000, 9000)
    return ev, dict(core=core, flank=flank, oldm=oldm, ba=ba, P=P)


EARTH = {2000: 0.0666, 3000: 0.0315, 4000: 0.0194, 5000: 0.0077}


def hyps(ev):
    s = {t: float((wgt * (ev >= t)).sum() / wgt.sum()) for t in (500, 1000, 2000, 3000, 4000, 5000)}
    return s, sum(abs(np.log(max(s[t], 1e-5) / EARTH[t])) for t in EARTH)


res = {'knobs': KN}
# Tune the amplitude and the back-arc uplift for hypsometry with the width variety on.
grid = []
for A in (4000.0, 4500.0, 5000.0, 5500.0):
    for U in (1000.0, 1500.0, 2000.0, 2500.0):
        for bs in (0.55, 0.7):
            ev, _ = model(A, KN['C'], KN['HW'], True, U, KN['BAW'], bs, KN['HILLU'])
            s, err = hyps(ev)
            grid.append((err, A, U, bs, s, float(ev[land].max())))
grid.sort(key=lambda g: g[0])
print('grid best (hypsometry, width variety and back-arc on):')
for g in grid[:6]:
    s = g[4]
    print(f"  A {g[1]:.0f} U {g[2]:.0f} strength {g[3]}: 0.5/1/2/3/4/5 km {s[500]:.3f}/{s[1000]:.3f}/{s[2000]:.3f}/{s[3000]:.3f}/{s[4000]:.3f}/{s[5000]:.4f} err {g[0]:.2f} max {g[5]:.0f}")
res['grid'] = [{'err': g[0], 'A': g[1], 'U': g[2], 'strength': g[3], 'above': {str(k): v for k, v in g[4].items()}, 'max': g[5]} for g in grid[:10]]
_, A, U, BS = grid[0][:4]
if os.environ.get('S80_FIX'):  # the chosen setting instead of the grid's best: 'A,U,strength'
    A, U, BS = (float(x) for x in os.environ['S80_FIX'].split(','))

variants = {
    'today (zone blobs, 7,000 m, plateau)': None,
    'SPIKE-79 plates (6,000 m, uniform width)': model(6000.0, 0.5, 90.0, False, 0.0, KN['BAW'], 1.0, 0.0),
    f'combined (A {A:.0f}, width variety, back-arc U {U:.0f}, old-margin hills)': model(A, KN['C'], KN['HW'], True, U, KN['BAW'], BS, KN['HILLU']),
}
today = np.clip(base + 7000 * ridge * zone * inland + hill + np.where(landall, 1200 * smooth(pn, 0.50, 0.62) * smooth(inland, 0.2, 0.6) * (1 - smooth(zone, 0.2, 0.6)), 0), -9000, 9000)
for name, v in variants.items():
    ev = today if v is None else v[0]
    s, err = hyps(ev)
    res.setdefault('variants', {})[name] = {'above': {str(k): x for k, x in s.items()}, 'err': err, 'max': float(ev[land].max())}
    print(f"{name:72s} | 0.5/1/2/3/4/5 km {s[500]:.3f}/{s[1000]:.3f}/{s[2000]:.3f}/{s[3000]:.3f}/{s[4000]:.3f}/{s[5000]:.4f} err {err:.2f} | max {ev[land].max():.0f}")
print('Earth without Antarctica: 2/3/4/5 km 0.067/0.032/0.019/0.008 (0.5/1 km with ice 0.467/0.272)')

ev, m = variants[list(variants)[-1]]
# The default class map. Memberships (smooth): high mountains, mountains, hills, plateau; the rest plain or rolling plain.
mt = smooth(m['core'], 0.25, 0.55)
hi = mt * smooth(ev, 2000.0, 3000.0)
hl = (1 - mt) * smooth(np.maximum(m['flank'], m['oldm']), 0.2, 0.45)
pl = (1 - mt) * (1 - hl) * smooth(np.maximum(m['P'], m['ba']), 0.3, 0.6)
rest = np.clip(1 - mt - hl - pl, 0, 1)
roll = rest * smooth(0.5 + 0.5 * fbm(P3, 3.0, 3, 8021), 0.45, 0.6)
plain = rest - roll
K = {'plain': 1.0, 'rolling plain': 1.0, 'hills': 5.0, 'mountains': 6.0, 'high mountains': 7.5, 'plateau': 1.0}
FLOOR = {'plain': 0.2, 'rolling plain': 0.35, 'hills': 0.0, 'mountains': 0.0, 'high mountains': 0.0, 'plateau': 0.0}
memb = {'plain': plain, 'rolling plain': roll, 'hills': hl, 'mountains': mt - hi, 'high mountains': hi, 'plateau': pl}
k = sum(K[n] * memb[n] for n in memb)
floor = sum(FLOOR[n] * memb[n] for n in memb)
cls_name = list(memb)
cls_idx = np.argmax(np.stack([memb[n] for n in cls_name], -1), -1)
share_m = {n: float((wgt * (cls_idx == j)).sum() / wgt.sum()) for j, n in enumerate(cls_name)}
print('class membership shares:', {n: round(v, 3) for n, v in share_m.items()})
# SPIKE-69's relief classes by k R14 (today's field relief at k 1 is the stand-in for the class's base relief; caveat in results).
rc = np.digitize(k * r14, [100, 300, 1000])
rc = np.where(ev >= 2500, np.maximum(rc, 2), rc)
sh = [float((wgt * (rc == j)).sum() / wgt.sum()) for j in range(4)]
print(f'relief classes k R14: plains {sh[0]:.3f} hills {sh[1]:.3f} mountains {sh[2]+sh[3]:.3f} (high {sh[3]:.3f}); Earth 0.61 / 0.14 / 0.25')
d = np.concatenate([np.abs(np.diff(k, axis=0))[land[:-1] & land[1:]], np.abs(np.diff(k, axis=1))[land[:, :-1] & land[:, 1:]]])
print(f'k step per 0.25 degree: p99 {np.percentile(d, 99):.2f} max {d.max():.2f}')
res['class_membership'] = share_m
res['relief_classes_kR14'] = {'plains': sh[0], 'hills': sh[1], 'mountains': sh[2] + sh[3], 'high': sh[3]}
res['dk'] = {'p99': float(np.percentile(d, 99)), 'max': float(d.max())}

# Outputs for the site runs: the delta to today's sample elevation and k, floor (f32, 1440 x 720, row 0 at 90 N).
np.stack([ev - e, k, floor], -1).astype('<f4').tofile(f'{out}/s80-fields.f32')
# Re-match the scorecard sites: per landform, the class and the real 0.5-degree window's mean height (SPIKE-55 matched the mean and spread per
# 1 degree; the coarse model has no fine spread, so the class stands in for it). Two sites at least 8 degrees apart, |lat| under 55.
WANT = {'N52E012': 'plain', 'N41W096': 'rolling plain', 'N43W102': 'plateau', 'N31W004': 'plateau', 'N40W078': 'hills', 'N48E008': 'hills',
        'N57W005': 'hills', 'N47E011': 'mountains', 'N27E086': 'high mountains'}
real = json.load(open(sys.argv[3])) if len(sys.argv) > 3 and sys.argv[3] != '-' else {}
# Window mean of the model over 0.5 degree (2 x 2 cells), and the class purity there.
def win(x):
    return 0.25 * (x + np.roll(x, -1, 0) + np.roll(x, -1, 1) + np.roll(np.roll(x, -1, 0), -1, 1))
evw = win(ev)
sites = {}
for tile, cname in WANT.items():
    j = cls_name.index(cname)
    pure = win((cls_idx == j).astype(float)) * win(land.astype(float))
    mean_real = real.get(tile, {}).get('mean', 0.0)
    # Stand-in limits (SPIKE-80, run 1): the site run adds the delta to today's stored field, so the window must be land well inland (no coast
    # inside 1 degree: a delta on land cells only makes a cliff at the coast) and, for hills and mountains, on today's ground at 300 m or more,
    # where today's crease amplitude is full (GEN-061's ramp); on lower ground today's field carries too little fine relief for k to scale.
    from scipy.ndimage import uniform_filter
    land9 = uniform_filter(landall.astype(float), size=9, mode='wrap')
    e_ok = (uniform_filter(e, size=3, mode='wrap') >= 300.0) if cname in ('hills', 'mountains', 'high mountains') else np.ones_like(e, bool)
    cost = np.abs(evw - mean_real) / max(mean_real, 200.0) + 2.0 * (1 - pure) + 10.0 * (np.abs(LAT) > 55) + 10.0 * (land9 < 0.99) + 10.0 * (~e_ok)
    picks = []
    for idx in np.argsort(cost, axis=None)[:200000]:
        r, cidx = np.unravel_index(idx, cost.shape)
        p = P3[r, cidx]
        if all(np.degrees(np.arccos(np.clip(p @ q, -1, 1))) >= 8.0 for q in [P3[a_, b_] for a_, b_ in picks]):
            picks.append((r, cidx))
        if len(picks) == 2:
            break
    sites[tile] = [{'lat': float(lat[r] - 0.125), 'lon': float(lon[cidx] + 0.125), 'mean_model': float(evw[r, cidx]), 'k': float(k[r, cidx]),
                    'floor': float(floor[r, cidx]), 'purity': float(pure[r, cidx])} for r, cidx in picks]
    print(tile, cname, 'real mean', mean_real, '->', [(round(s_['lat'], 2), round(s_['lon'], 2), round(s_['mean_model']), round(s_['k'], 2), round(s_['purity'], 2)) for s_ in sites[tile]])
res['sites'] = sites
json.dump(res, open(f'{out}/combined.json', 'w'), indent=1)

# Maps (0.25 degree, saved at half size): elevation 0-6 km (today | combined), the class map, k.
def ramp(x):
    t = np.clip(x / 6000.0, 0, 1)[..., None]
    lo_c, mid_c, hi_c = np.array([0.30, 0.55, 0.25]), np.array([0.65, 0.45, 0.25]), np.array([1.0, 1.0, 1.0])
    c = np.where(t < 0.5, lo_c + (mid_c - lo_c) * (t / 0.5), mid_c + (hi_c - mid_c) * ((t - 0.5) / 0.5))
    return np.where(landall[..., None], c, np.array([0.12, 0.2, 0.35]))
def save(img, path):
    Image.fromarray((np.clip(img, 0, 1) * 255).astype(np.uint8)).resize((img.shape[1] // 2, img.shape[0] // 2), Image.LANCZOS).save(path)
sep = np.ones((H, 8, 3))
save(np.concatenate([ramp(today), sep, ramp(ev)], 1), f'{out}/elevation-today-vs-combined.png')
COL = {'plain': [0.62, 0.78, 0.45], 'rolling plain': [0.80, 0.85, 0.50], 'hills': [0.66, 0.42, 0.25], 'mountains': [0.55, 0.50, 0.48],
       'high mountains': [0.95, 0.95, 0.95], 'plateau': [0.85, 0.62, 0.35]}
cimg = np.array([COL[n] for n in cls_name])[cls_idx]
cimg = np.where(landall[..., None], cimg, np.array([0.12, 0.2, 0.35]))
kimg = np.where(landall[..., None], (np.clip((k - 1) / 6.5, 0, 1)[..., None] * np.array([0.9, 0.3, 0.1]) + (1 - np.clip((k - 1) / 6.5, 0, 1))[..., None] * np.array([0.95, 0.92, 0.8])), np.array([0.12, 0.2, 0.35]))
save(np.concatenate([cimg, sep, kimg], 1), f'{out}/classes-and-k.png')
print(f'done {time.time() - t0:.0f} s; chosen A {A} U {U} strength {BS}')
