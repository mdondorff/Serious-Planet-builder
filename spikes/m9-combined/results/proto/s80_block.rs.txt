    // SPIKE-80 (spike code, never merged; from SPIKE-78): `S80_DIR=DIR S80_FIELDS=FILE` runs the combined generator at the sites of `--sites`
    // (re-matched by SPIKE-80's `combined.py`). FILE is 1440 x 720 x 3 f32 (row 0 at 89.875 N): the delta to today's sample elevation, the class map's
    // k and crease floor. The delta enters the fine and coarse windows before the pipeline (a stand-in for a re-baked store); k and the floor per cell
    // from the class map; then ADR 0088's two passes, thermal and deposition. Plateau landforms (N43W102, N31W004) also run their authored arid class:
    // badlands without a carve at k 1 (SPIKE-78), sand sea at k 0.35 with the carve at half strength (SPIKE-77).
    if let Ok(dir) = std::env::var("S80_DIR") {
        use planet_generators::height::HeightField as _;
        use std::sync::atomic::Ordering::Relaxed;
        let dir = PathBuf::from(dir);
        std::fs::create_dir_all(&dir).map_err(|e| e.to_string())?;
        let fields: Vec<f32> = std::fs::read(std::env::var("S80_FIELDS").map_err(|e| e.to_string())?)
            .map_err(|e| e.to_string())?
            .chunks_exact(4)
            .map(|b| f32::from_le_bytes([b[0], b[1], b[2], b[3]]))
            .collect();
        const FW: usize = 1440;
        const FH: usize = 720;
        assert_eq!(fields.len(), FW * FH * 3);
        // Bilinear lookup of channel `ch` at a unit direction.
        let look = |d: planet_core::Vec3, ch: usize| -> f64 {
            let lat = libm::asin(d.z.clamp(-1.0, 1.0)).to_degrees();
            let lon = libm::atan2(d.y, d.x).to_degrees();
            let fy = ((90.0 - lat) / 0.25 - 0.5).clamp(0.0, (FH - 2) as f64);
            let fx = (lon + 180.0) / 0.25 - 0.5;
            let (y0, ty) = (fy as usize, fy - fy.floor());
            let xf = fx.floor();
            let tx = fx - xf;
            let x0 = (xf as i64).rem_euclid(FW as i64) as usize;
            let x1 = (x0 + 1) % FW;
            let v = |y: usize, x: usize| f64::from(fields[(y * FW + x) * 3 + ch]);
            (v(y0, x0) * (1.0 - tx) + v(y0, x1) * tx) * (1.0 - ty) + (v(y0 + 1, x0) * (1.0 - tx) + v(y0 + 1, x1) * tx) * ty
        };
        let g = ts::GRID;
        let smooth = |x: f64, a: f64, b: f64| {
            let t = ((x - a) / (b - a)).clamp(0.0, 1.0);
            t * t * (3.0 - 2.0 * t)
        };
        let box_mean = |z: &[f64], n: usize, r: usize| -> Vec<f64> {
            let w = n + 1;
            let mut s = vec![0f64; w * w];
            for j in 0..n {
                let mut row = 0.0;
                for i in 0..n {
                    row += z[j * n + i];
                    s[(j + 1) * w + i + 1] = s[j * w + i + 1] + row;
                }
            }
            let mut out = vec![0f64; n * n];
            for j in 0..n {
                for i in 0..n {
                    let (i0, i1, j0, j1) = (i.saturating_sub(r), (i + r + 1).min(n), j.saturating_sub(r), (j + r + 1).min(n));
                    out[j * n + i] = (s[j1 * w + i1] - s[j0 * w + i1] - s[j1 * w + i0] + s[j0 * w + i0]) / ((i1 - i0) * (j1 - j0)) as f64;
                }
            }
            out
        };
        let frame = |lat: f64, lon: f64| {
            let (la, lo) = (lat.to_radians(), lon.to_radians());
            let (sla, cla, slo, clo) = (libm::sin(la), libm::cos(la), libm::sin(lo), libm::cos(lo));
            (
                planet_core::Vec3::new(cla * clo, cla * slo, sla),
                planet_core::Vec3::new(-slo, clo, 0.0),
                planet_core::Vec3::new(-sla * clo, -sla * slo, cla),
            )
        };
        const NC: usize = 491;
        const CC: f64 = 814.0;
        let report = |label: &str, s: &RasterStats, real: &RasterStats| -> String {
            let d = ts::distances(real, s);
            let cs: Vec<String> = ts::CONTINUOUS
                .iter()
                .enumerate()
                .map(|(k, c)| format!("{}={:.3}/{}", c.id, s.series[k].p(50), d.continuous[k].map_or("-".into(), |x| format!("{x:.2}"))))
                .collect();
            format!(
                "S80 {label} | flat {:.1}% | classes L1 {:.2} | density ratio {} | {}",
                100.0 * s.forms[0],
                d.forms_l1,
                d.density_ratio.map_or("-".into(), |x| format!("{x:.2}")),
                cs.join(" ")
            )
        };
        let half_m = ts::HALF_DEG.to_radians() * radius;
        for (i, s) in specs.iter().enumerate() {
            let (rs, _) = ts::analyse(&reals[i], g, g, threads, false);
            let cs: Vec<String> = ts::CONTINUOUS.iter().enumerate().map(|(k, c)| format!("{}={:.3}", c.id, rs.series[k].p(50))).collect();
            println!("S80 {} real | flat {:.1}% | {}", s.tile, 100.0 * rs.forms[0], cs.join(" "));
            for (k, &(lat, lon)) in s.sites.iter().enumerate() {
                let z0 = &gens[i][k];
                let (bs, _) = ts::analyse(z0, g, g, threads, false);
                println!("{}", report(&format!("{} site{} today-at-new-site", s.tile, k + 1), &bs, &rs));
                let dirs = ts::generator_directions(lat, lon, radius);
                planet_generators::erosion::spike73_set_floor(0.35);
                let z1 = field_s74.heights_threaded(&dirs, &threads);
                let (c, east, north) = frame(lat, lon);
                let mut dc = Vec::with_capacity(NC * NC);
                for row in 0..NC {
                    for col in 0..NC {
                        let (x, y) = ((col as f64 - (NC / 2) as f64) * CC, ((NC / 2) as f64 - row as f64) * CC);
                        let p = c * radius + east * x + north * y;
                        dc.push(p * (1.0 / libm::sqrt(p.x * p.x + p.y * p.y + p.z * p.z)));
                    }
                }
                let zc = field_s74.heights_threaded(&dc, &threads);
                planet_generators::erosion::spike73_set_floor(0.0);
                let df: Vec<[f64; 3]> = dirs.iter().map(|d| [look(*d, 0), look(*d, 1), look(*d, 2)]).collect();
                let dcf3: Vec<[f64; 3]> = dc.iter().map(|d| [look(*d, 0), look(*d, 1), look(*d, 2)]).collect();
                {
                    let mut ks: Vec<f64> = df.iter().map(|v| v[1]).collect();
                    ks.sort_by(f64::total_cmp);
                    let mut ds: Vec<f64> = df.iter().map(|v| v[0]).collect();
                    ds.sort_by(f64::total_cmp);
                    let m1 = z1.iter().sum::<f64>() / z1.len() as f64;
                    println!(
                        "S80SITE {} site{} ({lat}, {lon}) | window mean today {:.0} m | delta p10/p50/p90 {:.0}/{:.0}/{:.0} m | k p10/p50/p90 {:.2}/{:.2}/{:.2} | floor p50 {:.2}",
                        s.tile,
                        k + 1,
                        m1,
                        ds[ds.len() / 10],
                        ds[ds.len() / 2],
                        ds[ds.len() * 9 / 10],
                        ks[ks.len() / 10],
                        ks[ks.len() / 2],
                        ks[ks.len() * 9 / 10],
                        df[df.len() / 2][2]
                    );
                }
                let plateau = matches!(s.tile.as_str(), "N43W102" | "N31W004");
                let names: &[&str] = if plateau { &["combined", "combined+authored"] } else { &["combined"] };
                for &name in names {
                    let t = Instant::now();
                    // (k override, carve strength in %, carve on)
                    let (kov, kdt, carve): (Option<f64>, u32, bool) = match (name, s.tile.as_str()) {
                        ("combined+authored", "N43W102") => (Some(1.0), 100, false),
                        ("combined+authored", _) => (Some(0.35), 50, true),
                        _ => (None, 100, true),
                    };
                    let zb: Vec<f64> = (0..g * g)
                        .map(|p| {
                            let fl = if kov.is_some() { 0.0 } else { df[p][2] };
                            z0[p] + (z1[p] - z0[p]) * (fl / 0.35) + if z1[p] > 0.0 { df[p][0] } else { 0.0 }
                        })
                        .collect();
                    let zcb: Vec<f64> = (0..NC * NC).map(|q| zc[q] + if zc[q] > 0.0 { dcf3[q][0] } else { 0.0 }).collect();
                    let kk = |kc: f64, e: f64| 1.0 + smooth(e, 0.0, 400.0) * (kov.unwrap_or(kc) - 1.0);
                    let mf = box_mean(&zb, g, 167);
                    let h: Vec<f64> = (0..g * g).map(|p| if zb[p] > 0.0 { mf[p] + kk(df[p][1], mf[p]) * (zb[p] - mf[p]) } else { zb[p] }).collect();
                    let mut h2: Vec<f32> = h.iter().map(|x| *x as f32).collect();
                    k74::KDT_PCT.store(kdt, Relaxed);
                    if carve {
                        let at_coarse = |a: &[f64], row: usize, col: usize| -> f64 {
                            let (x, y) = (-half_m + (col as f64 + 0.5) * ts::CELL_M, half_m - (row as f64 + 0.5) * ts::CELL_M);
                            let fx = (x / CC + (NC / 2) as f64).clamp(0.0, (NC - 2) as f64);
                            let fy = ((NC / 2) as f64 - y / CC).clamp(0.0, (NC - 2) as f64);
                            let (x0, y0) = (fx as usize, fy as usize);
                            let (tx, ty) = (fx - x0 as f64, fy - y0 as f64);
                            (a[y0 * NC + x0] * (1.0 - tx) + a[y0 * NC + x0 + 1] * tx) * (1.0 - ty) + (a[(y0 + 1) * NC + x0] * (1.0 - tx) + a[(y0 + 1) * NC + x0 + 1] * tx) * ty
                        };
                        let mcs = box_mean(&zcb, NC, 6);
                        let hc: Vec<f32> =
                            (0..NC * NC).map(|q| if zcb[q] > 0.0 { (mcs[q] + kk(dcf3[q][1], mcs[q]) * (zcb[q] - mcs[q])) as f32 } else { zcb[q] as f32 }).collect();
                        let mut cc = hc.clone();
                        let gc = k74::G { w: NC, ox: 0, oy: 0, fix: &[], dx: CC, frec: &[], farea: &[], amax: 0.0 };
                        k74::UK.store(20, Relaxed);
                        k74::LOCAL.store(0, Relaxed);
                        k74::carve(gc, &mut cc, &[]);
                        let dcv: Vec<f64> = (0..NC * NC).map(|q| f64::from(cc[q] - hc[q])).collect();
                        for row in 0..g {
                            for col in 0..g {
                                let p = row * g + col;
                                if zb[p] > 0.0 {
                                    h2[p] = (h[p] + at_coarse(&dcv, row, col)) as f32;
                                }
                            }
                        }
                    }
                    let orig: Vec<f32> = h.iter().map(|x| *x as f32).collect();
                    let g2 = k74::G { w: g, ox: 0, oy: 0, fix: &[], dx: ts::CELL_M, frec: &[], farea: &[], amax: 1e6 };
                    if carve {
                        k74::LOCAL.store(2, Relaxed);
                        k74::UK.store(0, Relaxed);
                        k74::carve(g2, &mut h2, &[]);
                    }
                    k74::thermal_active(g2, &mut h2, 33.0, 100, threads);
                    k74::deposit(g2, &mut h2, &orig, 1e5, 0.5);
                    k74::LOCAL.store(0, Relaxed);
                    k74::UK.store(20, Relaxed);
                    k74::KDT_PCT.store(100, Relaxed);
                    let hf: Vec<f64> = h2.iter().map(|x| f64::from(*x)).collect();
                    let (ps, _) = ts::analyse(&hf, g, g, threads, false);
                    println!("{}", report(&format!("{} site{} {}", s.tile, k + 1, name), &ps, &rs));
                    if k == 0 {
                        let b: Vec<u8> = hf.iter().flat_map(|x| (*x as f32).to_le_bytes()).collect();
                        std::fs::write(dir.join(format!("{}-site1-{}.f32", s.tile, name.replace('+', "-"))), b).map_err(|e| e.to_string())?;
                    }
                    eprintln!("S80 {} site{} {}: {:.0} s", s.tile, k + 1, name, t.elapsed().as_secs_f64());
                }
            }
        }
        return Ok(());
    }
